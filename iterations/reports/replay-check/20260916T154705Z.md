#!/usr/bin/env python3
"""W81: the consumer half of W80's open question.

W80 proved the buildings/creeps/wards streams' EMISSION ORDER changes between
dumps of the same .dem while their content multiset does not. It did NOT check
whether anyone downstream reduces those streams order-dependently.

This probe answers both halves on one corpus:
  (A) corpus  -- do duplicate keys exist, and do the siblings DISAGREE on the
                 fields consumers actually read?
  (B) consumer-- run the three real reducers found in the tree, verbatim, on
                 each dump and see whether they return different answers.
"""
import json, sys, collections

RADIANT, DIRE = 2, 3


# ---- the three real reducers, copied VERBATIM from the tree -----------------
def ancients_lanekill(tl):                       # lanekill_commit.py:112-118
    out = {}
    for b in tl['buildings']:
        if b.get('name') == 'ancient' and b['team'] not in out:
            out[b['team']] = (b['x'], b['y'])
    return out


def own_ancients_capmono(tl):                    # capmono_refusal.py:217-224
    a = {}
    for b in tl['buildings']:
        if b['name'] == 'ancient':
            a[b['team']] = (b['x'], b['y'])
        if len(a) == 2:
            break
    return a


def ancient_slotpush(tl):                        # slotpush_domain.py:275-289
    anc = {}
    for b in tl['buildings']:
        team = b.get('team')
        if team not in (RADIANT, DIRE):
            continue
        if b.get('name') == 'ancient':
            anc[team] = (b['x'], b['y'])
    return anc


def towers_stayfield(tl):                        # stayfield_domain.py:314-323
    seen = collections.defaultdict(list)
    for b in tl.get('buildings', ()):
        if b.get('name') != 'tower':
            continue
        seen[(round(b['x']), round(b['y']))].append(b)
        towers = {}
    towers = {}
    for key, rows in seen.items():
        rows.sort(key=lambda b: b['t'])          # stable
        towers[key] = rows[0]['team']
    return towers


def map_life_slotpush(tl):                       # slotpush_domain.py:284-287
    """(t, alive) samples per tower key -- the pair that decides tower lifespan."""
    samples = collections.defaultdict(dict)
    for b in tl['buildings']:
        team = b.get('team')
        if team not in (RADIANT, DIRE):

                (b['t'], bool(b.get('alive', True))))

#!/usr/bin/env python3
"""W81 follow-up: probe.py flagged slotpush Map.life as MOVES. That reading
compared RAW DICT ITERATION ORDER, which is not what the real class produces
(`self.life[(team,p)] = sorted(towers[p])`, `order = sorted(towers, key=dist)`).

This rebuilds the real fields -- life / hg / t2 -- and compares THOSE.
Also checks the tie case the stable sort would expose, and counts tower keys
per team (the ghost-tower question: the class asserts exactly 11).
"""
import json, sys, math, collections

RADIANT, DIRE = 2, 3
N_HIGH_GROUND, N_SECOND_TIER = 5, 3


def dist(x1, y1, x2, y2):
    return math.hypot(x1 - x2, y1 - y2)


def build_map_fields(tl):
    """slotpush_domain.Map.__init__, the parts that become state."""
    ancient = {}
    samples = collections.defaultdict(dict)
    for b in tl['buildings']:
        team = b.get('team')
        if team not in (RADIANT, DIRE):
            continue
        if b.get('name') == 'ancient':
            ancient[team] = (b['x'], b['y'])
        elif b.get('name') == 'tower':
            key = (round(b['x'], 1), round(b['y'], 1))
            samples[team].setdefault(key, []).append(
                (b['t'], bool(b.get('alive', True))))
    hg, t2, life, ntow, ties = {}, {}, {}, {}, {}
    for team in (RADIANT, DIRE):
        towers = samples.get(team, {})
        ntow[team] = len(towers)
        own = ancient[team]
        order = sorted(towers, key=lambda p: dist(p[0], p[1], own[0], own[1]))
        d = [dist(p[0], p[1], own[0], own[1]) for p in order]
        # would a tie let the stable sort leak insertion order into the tiers?
        ties[team] = sum(1 for i in range(1, len(d)) if abs(d[i] - d[i - 1]) < 1e-9)
        hg[team] = order[:N_HIGH_GROUND]
        t2[team] = order[N_HIGH_GROUND:N_HIGH_GROUND + N_SECOND_TIER]
        for p in order:
            life[(team, p)] = sorted(towers[p])
    return {'ancient': ancient, 'hg': hg, 't2': t2, 'life': life,
            'n_towers': ntow, 'exact_dist_ties': ties}


def canon(f):
    return json.dumps({
        'ancient': sorted(f['ancient'].items()),
        'hg': {str(k): v for k, v in sorted(f['hg'].items())},
        't2': {str(k): v for k, v in sorted(f['t2'].items())},
        'life': sorted((str(k), v) for k, v in f['life'].items()),
    }, sort_keys=True, default=list)


def main(paths):
    fields = []
    for p in paths:
        with open(p) as fh:
            tl = json.load(fh)
        fields.append(build_map_fields(tl))

    print('tower keys per team (class asserts exactly 11):')
    for i, f in enumerate(fields):
        print('  dump%d: %s   exact-distance ties (would leak order): %s'
              % (i + 1, f['n_towers'], f['exact_dist_ties']))

    cs = [canon(f) for f in fields]
    print()
    print('slotpush Map state (ancient + hg + t2 + life), canonicalised the way'
          ' the class itself builds it:')
    print('  identical across %d dumps: %s' % (len(cs), len(set(cs)) == 1))
    if len(set(cs)) != 1:
        for i, c in enumerate(cs):
            print('  dump%d md5-ish len=%d head=%s' % (i + 1, len(c), c[:160]))
        # locate the first differing life key
        a, b = fields[0]['life'], fields[1]['life']
        for k in sorted(set(a) | set(b), key=str):
            if a.get(k) != b.get(k):
                print('  first differing life key %r' % (k,))
                print('    dump1: %s' % (a.get(k),))
                print('    dump2: %s' % (b.get(k),))
                break
    return 0


if __name__ == '__main__':
    sys.exit(main(sys.argv[1:]))

#!/usr/bin/env python3
"""W81 reconciliation: W80 reported buildings/creeps/wards "first record of a key
changes 98.8%/100%/98.8% across dumps". W81 measures ZERO duplicate keys on
those streams. Both cannot be describing the same key.

Hypothesis: W80's key was COARSE (time-only, or name+time) -- under which many
DISTINCT entities share a key, so "which is first" varies without any entity
being duplicated. On `snapshots` the key (hero,t) DOES identify one entity, so
the same statistic there means something completely different.

This measures the same statistic under both keys and prints them side by side.
"""
import json, sys, collections

KEYS = {
    'coarse  t only': lambda r: (round(r['t'], 1),),
    'coarse  name+t': lambda r: (r.get('name'), round(r['t'], 1)),
    'entity  name+x+y+t': lambda r: (r.get('name'), round(r['x'], 1),
                                     round(r['y'], 1), round(r['t'], 1)),
}
CREEP_KEYS = {
    'coarse  t only': lambda r: (round(r['t'], 1),),
    'coarse  team+t': lambda r: (r.get('team'), round(r['t'], 1)),
    'entity  team+x+y+t': lambda r: (r.get('team'), round(r['x'], 1),
                                     round(r['y'], 1), round(r['t'], 1)),
}
SNAP_KEYS = {
    'entity? hero+t': lambda r: (r.get('hero'), round(r['t'], 1)),
    'entity  hero+idx+t': lambda r: (r.get('hero'), r.get('idx'),
                                     round(r['t'], 1)),
}


def first_of_each(rows, keyf):
    d = {}
    for r in rows:
        k = keyf(r)
        if k not in d:
            d[k] = r
    return d


def payload(r, stream):
    if stream == 'snapshots':
        return (r.get('hero'), r.get('idx'), r.get('hp'),
                round(r.get('x', 0), 1), round(r.get('y', 0), 1))
    return tuple(sorted((k, v) for k, v in r.items()))


def main(paths):
    dumps = []
    for p in paths:
        with open(p) as fh:
            dumps.append(json.load(fh))

    for stream, keyset in (('buildings', KEYS), ('creeps', CREEP_KEYS),
                           ('snapshots', SNAP_KEYS)):
        print('=' * 74)
        print(stream)
        print('=' * 74)
        rows0 = dumps[0].get(stream) or []
        # content multiset identical across dumps?
        multisets = [collections.Counter(payload(r, stream)
                                         for r in (d.get(stream) or []))
                     for d in dumps]
        print('  records=%d   content multiset identical across %d dumps: %s'
              % (len(rows0), len(dumps),
                 all(m == multisets[0] for m in multisets)))
        for label, keyf in keyset.items():
            groups = collections.defaultdict(int)
            for r in rows0:
                groups[keyf(r)] += 1
            n_keys = len(groups)
            n_dup = sum(1 for v in groups.values() if v > 1)
            firsts = [first_of_each(d.get(stream) or [], keyf) for d in dumps]
            shared = set(firsts[0])
            changed = sum(
                1 for k in shared
                if any(payload(firsts[i][k], stream)
                       != payload(firsts[0][k], stream)
                       for i in range(1, len(firsts))))
            dup_keys = [k for k, v in groups.items() if v > 1]
            denom = len(dup_keys) or 1
            changed_dup = sum(
                1 for k in dup_keys
                if any(payload(firsts[i][k], stream)
                       != payload(firsts[0][k], stream)
                       for i in range(1, len(firsts))))
            print('    %-20s keys=%-7d dup_keys=%-7d (%5.1f%%)   '
                  '"first record" differs on %d/%d dup keys = %5.1f%%'
                  % (label, n_keys, n_dup, 100.0 * n_dup / max(1, n_keys),
                     changed_dup, len(dup_keys), 100.0 * changed_dup / denom))
        print()
    return 0


if __name__ == '__main__':
    sys.exit(main(sys.argv[1:]))
